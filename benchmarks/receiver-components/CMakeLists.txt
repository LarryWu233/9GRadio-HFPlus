cmake_minimum_required(VERSION 3.22.1)
project(receiver_component_evaluation C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
foreach(source_dir SPEEX_SOURCE_DIR PIPEWIRE_DLL_DIR)
    if(NOT IS_DIRECTORY "${${source_dir}}")
        message(FATAL_ERROR "Provide ${source_dir}; see README.md for the pinned sources")
    endif()
endforeach()

# Standalone resampler, exactly the upstream OUTSIDE_SPEEX integration.
add_library(speex_eval STATIC "${SPEEX_SOURCE_DIR}/libspeexdsp/resample.c")
target_include_directories(speex_eval PUBLIC "${SPEEX_SOURCE_DIR}/include/speex"
    PRIVATE "${SPEEX_SOURCE_DIR}/libspeexdsp")
target_compile_definitions(speex_eval PUBLIC OUTSIDE_SPEEX FLOATING_POINT RANDOM_PREFIX=ninegradio_eval)
target_compile_options(speex_eval PRIVATE -O3)
if(ANDROID_ABI MATCHES "^(arm64-v8a|armeabi-v7a)$")
    target_compile_definitions(speex_eval PRIVATE USE_NEON)
elseif(ANDROID_ABI STREQUAL "x86_64")
    target_compile_definitions(speex_eval PRIVATE USE_SSE)
endif()

# The standalone DLL header does not use defs.h. Remove only that include;
# preserve the original controller, attribution and licence verbatim.
file(READ "${PIPEWIRE_DLL_DIR}/dll.h" DLL_HEADER)
string(REPLACE "#include <spa/utils/defs.h>" "/* Unused defs.h omitted for standalone evaluation. */" DLL_HEADER "${DLL_HEADER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dll.h" "${DLL_HEADER}")
add_executable(speex_clock_evaluation speex_clock_evaluation.c)
target_include_directories(speex_clock_evaluation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(speex_clock_evaluation speex_eval m)
target_compile_options(speex_clock_evaluation PRIVATE -O3)

if(IS_DIRECTORY "${LIQUID_SOURCE_DIR}")
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_AUTOTESTS OFF CACHE BOOL "" FORCE)
    set(BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
    set(FIND_FFTW OFF CACHE BOOL "" FORCE)
    set(FIND_SIMD OFF CACHE BOOL "" FORCE)
    set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
    set(ENABLE_COLOR OFF CACHE BOOL "" FORCE)
    set(ENABLE_TIMESTAMPS OFF CACHE BOOL "" FORCE)
    if(ANDROID_ABI MATCHES "^(arm64-v8a|armeabi-v7a)$")
        set(C_NEON_FOUND TRUE)
    endif()
    add_subdirectory("${LIQUID_SOURCE_DIR}" liquid EXCLUDE_FROM_ALL)
    add_executable(liquid_iq_evaluation liquid_iq_evaluation.c)
    target_link_libraries(liquid_iq_evaluation liquid-static m)
    target_compile_options(liquid_iq_evaluation PRIVATE -O3)
endif()
